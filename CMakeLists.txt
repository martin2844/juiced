cmake_minimum_required(VERSION 3.24)
project(JuicedMac VERSION 0.3.3 LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "This target requires macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3 URL https://github.com/libsdl-org/SDL/archive/829a65d769d935c4852f8159e964312c0957260a.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(SDL3)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "")
set(CMAKE_DISABLE_FIND_PACKAGE_SDL2 ON CACHE BOOL "" FORCE)
set(PROFILE_METAL ON CACHE BOOL "" FORCE)
FetchContent_Declare(mojoshader URL https://github.com/icculus/mojoshader/archive/ad5dff84830c2863c841f4b1f4e3df78c705b383.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(mojoshader)
target_compile_definitions(mojoshader PRIVATE USE_SDL3)
target_link_libraries(mojoshader PUBLIC SDL3::SDL3)
file(GLOB_RECURSE CORE CONFIGURE_DEPENDS src/Core/*.cpp)
list(FILTER CORE EXCLUDE REGEX "/(CpuInfo|Cpu|Config|Text|Log|GameSpyVoice)\\.cpp$")
add_library(juice_core STATIC ${CORE} src/Mac/Base.cpp src/Mac/Compat/Windows.cpp src/Mac/VoiceUnavailable.cpp)
target_include_directories(juice_core PUBLIC src src/Mac/Compat)
target_compile_options(juice_core PUBLIC -include ${CMAKE_CURRENT_SOURCE_DIR}/src/Mac/Compat/Crt.h)
target_compile_definitions(juice_core PUBLIC _DEBUG OJ_MACOS)
target_link_libraries(juice_core PUBLIC SDL3::SDL3)

file(GLOB_RECURSE RENDER CONFIGURE_DEPENDS src/Render/*.cpp)
list(FILTER RENDER EXCLUDE REGEX "D3D9/|RenderDevice\\.cpp$|HlslCompiler\\.cpp$")
add_library(juice_render STATIC ${RENDER} src/Mac/MetalDevice.cpp src/Mac/Shader.cpp)
target_include_directories(juice_render PRIVATE ${mojoshader_SOURCE_DIR})
target_link_libraries(juice_render PUBLIC juice_core mojoshader)

find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libswresample)
add_library(juice_audio STATIC src/Audio/Bank.cpp src/Audio/Player.cpp src/Audio/Voices.cpp src/Mac/Audio.cpp src/Mac/Streams.cpp)
target_link_libraries(juice_audio PUBLIC juice_core PkgConfig::FFMPEG)
file(GLOB_RECURSE GAME CONFIGURE_DEPENDS src/Game/*.cpp)
list(FILTER GAME EXCLUDE REGEX "App/(Main|CrashLog)\\.cpp$|Platform/(Platform|InputDevices|Settings|SystemKeys|Win7Imports|Windows7)\\.cpp$|Media/Binkw32\\.cpp$")
add_library(juice_game STATIC ${GAME} src/Mac/Platform.cpp src/Mac/Controls.cpp src/Mac/ControlsOverlay.cpp src/Mods/Runtime.cpp src/Mods/Creative.cpp src/Mods/CreativeMenu.cpp src/Mods/ImageImport.mm)
target_link_libraries(juice_game PUBLIC juice_render juice_audio "-framework AppKit" "-framework ImageIO")
find_package(Lua 5.4 REQUIRED)
target_include_directories(juice_game PRIVATE ${LUA_INCLUDE_DIR})
# Static runtime: users need neither Homebrew nor a Lua installation.
find_library(JUICED_LUA_STATIC NAMES liblua.a HINTS /opt/homebrew/lib REQUIRED)
target_link_libraries(juice_game PUBLIC ${JUICED_LUA_STATIC})
add_executable(juiced src/Mac/Main.cpp src/Mac/Launcher.mm)
target_link_libraries(juiced PRIVATE juice_game "-framework AppKit")
add_executable(juiced_app MACOSX_BUNDLE src/Mac/Main.cpp src/Mac/Launcher.mm)
target_link_libraries(juiced_app PRIVATE juice_game "-framework AppKit")
set_target_properties(juiced_app PROPERTIES
  OUTPUT_NAME "Juiced"
  MACOSX_BUNDLE_GUI_IDENTIFIER "org.openjuice.macos"
  MACOSX_BUNDLE_BUNDLE_NAME "Juiced"
  MACOSX_BUNDLE_BUNDLE_VERSION "0.3.3"
  MACOSX_BUNDLE_SHORT_VERSION_STRING "0.3.3")

include(CTest)
if(BUILD_TESTING)
  file(GLOB_RECURSE TESTS CONFIGURE_DEPENDS tests/Unit/*.cpp)
  list(FILTER TESTS EXCLUDE REGEX "CpuInfoTests\\.cpp$|VoiceTests\\.cpp$|SystemCommandTests\\.cpp$|DirectXCheckTests\\.cpp$")
  file(GLOB_RECURSE INTEGRATION CONFIGURE_DEPENDS
    tests/Integration/GameData/*.cpp tests/Integration/Frontend/*.cpp
    tests/Integration/Race/*.cpp tests/Integration/Saves/*.cpp
    tests/Integration/Cars/*.cpp tests/Integration/Audio/SoundListenerTests.cpp tests/Integration/Audio/VoiceManagerTests.cpp)
  # Binary-comparison tests require the original Windows executable, not just game assets.
  list(FILTER INTEGRATION EXCLUDE REGEX "ExeTablesTests\\.cpp$")
  add_executable(juiced_tests ${TESTS} ${INTEGRATION} tests/Framework/TestMain.cpp tests/Framework/Fixtures.cpp src/Mac/TestMain.cpp src/JuicedConfig/ConfigData.cpp)
  target_compile_definitions(juiced_tests PRIVATE JUICED_TEST_MODS="${CMAKE_CURRENT_SOURCE_DIR}/mods")
  target_include_directories(juiced_tests PRIVATE tests)
  target_link_libraries(juiced_tests PRIVATE juice_game)
  add_executable(mac_car_mod_check tests/Mac/CarModTest.cpp)
  target_include_directories(mac_car_mod_check PRIVATE tests)
  target_link_libraries(mac_car_mod_check PRIVATE juice_game)
  add_test(NAME core_regression COMMAND juiced_tests --unit)
endif()

add_executable(juiced_shader_check EXCLUDE_FROM_ALL tools/ShaderCheck.cpp)
target_include_directories(juiced_shader_check PRIVATE ${mojoshader_SOURCE_DIR})
target_link_libraries(juiced_shader_check PRIVATE juice_render)

if(BUILD_TESTING)
  add_executable(mac_controls_check tests/Mac/ControlsTest.cpp)
  target_link_libraries(mac_controls_check PRIVATE juice_game)
  add_test(NAME mac_controls COMMAND mac_controls_check)
  add_executable(mac_controls_overlay_check tests/Mac/ControlsOverlayTest.cpp)
  target_link_libraries(mac_controls_overlay_check PRIVATE juice_game)
  add_test(NAME mac_controls_overlay COMMAND mac_controls_overlay_check)
endif()

if(BUILD_TESTING)
  add_executable(mac_reflection_check tests/Mac/ReflectionTest.cpp)
  target_link_libraries(mac_reflection_check PRIVATE juice_render)
  add_test(NAME mac_reflections COMMAND mac_reflection_check)
endif()

if(BUILD_TESTING)
  add_executable(mac_creative_check tests/Mac/CreativeTest.cpp)
  target_include_directories(mac_creative_check PRIVATE tests)
  target_link_libraries(mac_creative_check PRIVATE juice_game)
  add_test(NAME mac_creative COMMAND mac_creative_check)
  target_compile_definitions(mac_creative_check PRIVATE JUICED_TEST_MODS="${CMAKE_CURRENT_SOURCE_DIR}/mods")
endif()

if(BUILD_TESTING)
  add_executable(mac_mod_runtime_check tests/Mac/RuntimeTest.cpp)
  target_link_libraries(mac_mod_runtime_check PRIVATE juice_game)
  add_test(NAME mac_mod_runtime COMMAND mac_mod_runtime_check)
  add_executable(mac_neon_picker_check tests/Mac/NeonPickerTest.cpp)
  target_include_directories(mac_neon_picker_check PRIVATE tests)
  target_link_libraries(mac_neon_picker_check PRIVATE juice_game)
  target_compile_definitions(mac_neon_picker_check PRIVATE JUICED_TEST_MODS="${CMAKE_CURRENT_SOURCE_DIR}/mods")
endif()
